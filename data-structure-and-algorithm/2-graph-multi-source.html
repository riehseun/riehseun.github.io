Rotting Oranges (150/250)

class Solution:
    def orangesRotting(self, grid: List[List[int]]) -> int:
        
        # BFS multi-source
        # From each rotten fruit, compute distance to nearby nodes
        # Return max distance
        # Time O(nm) n = len(grid), m = len(grid[0])
        # Space O(nm)

        n, m = len(grid), len(grid[0])
        dirs = [(-1,0), (1,0), (0,1), (0,-1)]
        queue = deque()
        depth = 0

        for i in range(n):
            for j in range(m):
                if grid[i][j] == 2:
                    queue.append((i,j,0))

        while queue:
            i, j, d = queue.popleft()
            for x, y in dirs:
                nx, ny = i + x, j + y
                if 0 <= nx < n \
                    and 0 <= ny < m \
                    and grid[nx][ny] == 1:
                    queue.append((nx, ny, d+1))
                    grid[nx][ny] = 2
            depth = d

        for i in range(n):
            for j in range(m):
                if grid[i][j] == 1:
                    return -1

        return depth

Islands and Treasure

class Solution:
    def islandsAndTreasure(self, grid: List[List[int]]) -> None:
        
        # BFS multi-source
        # From each treasure chest, compute distance to nearby node
        # Use min distance if more than one distance is computed
        # Time O(nm) n = len(grid), m = len(grid[0])
        # Space O(nm)

        n, m = len(grid), len(grid[0])
        dirs = [(-1,0), (1,0), (0,-1), (0,1)]
        queue = deque()

        for i in range(n):
            for j in range(m):
                if grid[i][j] == 0:
                    queue.append((i,j))

        while queue:
            i, j = queue.popleft()

            for x, y in dirs:
                nx, ny = i + x, j + y
                if 0 <= nx < n \
                    and 0 <= ny < m \
                    and grid[nx][ny] == 2**31 - 1:
                    queue.append((nx, ny))
                    grid[nx][ny] = grid[i][j] + 1
