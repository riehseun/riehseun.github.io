Single Neuron

import numpy as np
from numpy.typing import NDArray


class Solution:
    def forward(self, x: NDArray[np.float64], w: NDArray[np.float64], b: float, activation: str) -> float:
        # x: 1D input array
        # w: 1D weight array (same length as x)
        # b: scalar bias
        # activation: "sigmoid" or "relu"
        #
        # Pre-activation: z = dot(x, w) + b
        # Sigmoid: σ(z) = 1 / (1 + exp(-z))
        # ReLU: max(0, z)
        # return round(your_answer, 5)
        z = np.dot(x, w) + b
        if activation == "relu":
            z = np.maximum(0.0, z)
        if activation == "sigmoid":
            z = 1 / (1 + np.exp(-z))
        return np.round(z, 5)

Backpropagation

import numpy as np
from numpy.typing import NDArray
from typing import Tuple


class Solution:
    def backward(self, x: NDArray[np.float64], w: NDArray[np.float64], b: float, y_true: float) -> Tuple[NDArray[np.float64], float]:
        # x: 1D input array
        # w: 1D weight array
        # b: scalar bias
        # y_true: true target value
        #
        # Forward: z = dot(x, w) + b, y_hat = sigmoid(z)
        # Loss: L = 0.5 * (y_hat - y_true)^2
        # Return: (dL_dw rounded to 5 decimals, dL_db rounded to 5 decimals)
        z = np.dot(x, w) + b
        y_hat = 1 / (1+np.exp(-z))
        loss = 0.5 * (y_hat-y_true)**2
        dw = (y_hat-y_true)*y_hat*(1-y_hat)*x
        db = (y_hat-y_true)*y_hat*(1-y_hat)
        return (np.round(dw,5), np.round(db,5))

Multi-Layer Backpropagation

import numpy as np
from typing import List


class Solution:
    def forward_and_backward(self,
                              x: List[float],
                              W1: List[List[float]], b1: List[float],
                              W2: List[List[float]], b2: List[float],
                              y_true: List[float]) -> dict:
        # Architecture: x -> Linear(W1, b1) -> ReLU -> Linear(W2, b2) -> predictions
        # Loss: MSE = mean((predictions - y_true)^2)
        #
        # Return dict with keys:
        #   'loss':  float (MSE loss, rounded to 4 decimals)
        #   'dW1':   2D list (gradient w.r.t. W1, rounded to 4 decimals)
        #   'db1':   1D list (gradient w.r.t. b1, rounded to 4 decimals)
        #   'dW2':   2D list (gradient w.r.t. W2, rounded to 4 decimals)
        #   'db2':   1D list (gradient w.r.t. b2, rounded to 4 decimals)
        x, W1, b1, W2, b2, y_true = np.array(x), np.array(W1), np.array(b1), np.array(W2), np.array(b2), np.array(y_true)
        z1 = np.dot(x, W1.T) + b1
        a1 = np.maximum(0, z1)
        z2 = np.dot(a1, W2.T) + b2
        loss = np.mean((z2-y_true)**2)
        dz2 = (2) * (z2-y_true)
        dW2 = a1 * dz2
        db2 = dz2
        da1 = dz2 * W2
        dz1 = da1 * (z1>0)
        dW1 = x * dz1.T
        db1 = dz1
        dW1[dW1 == -0.0] = 0.0

        return {
            "loss": round(loss, 4),
            "dW1": np.round(dW1, 4).tolist(),
            "db1": np.round(db1.flatten(), 4).tolist(),
            "dW2": np.round(dW2.reshape(1, -1), 4).tolist(),
            "db2": np.round(db2.flatten(), 4).tolist(),
        }

MLP from Scratch

import numpy as np
from numpy.typing import NDArray
from typing import List


class Solution:
    def forward(self, x: NDArray[np.float64], weights: List[NDArray[np.float64]], biases: List[NDArray[np.float64]]) -> NDArray[np.float64]:
        # x: 1D input array
        # weights: list of 2D weight matrices
        # biases: list of 1D bias vectors
        # Apply ReLU after each hidden layer, no activation on output layer
        # return np.round(your_answer, 5)

        num_layers, a = len(weights), x
        for l in range(num_layers):
            z = np.dot(a, weights[l]) + biases[l]
            a = np.maximum(0, z)
        return np.round(z, 5)
